# Pinned to the versions this was developed and tested against.
#
# numpy is pinned explicitly even though nothing here imports it directly.
# pandas 1.5.3 ships compiled extensions built against the numpy 1.x C API and
# fails at import with "numpy.dtype size changed" under numpy 2.x. Leaving numpy
# unpinned lets any later install resolve it to 2.x and break pandas, which is
# exactly what happened to the global interpreter on this machine.
numpy==1.23.5

boto3==1.34.162
botocore==1.34.162
requests==2.28.2
pandas==1.5.3

# PDF text extraction. pypdfium2 is the text backend: this regulation uses
# embedded fonts with no usable ToUnicode mapping, so pdfminer (under pdfplumber)
# emits (cid:NNN) markers in place of characters. pdfplumber is kept only for
# extract_tables, since PDFium returns text without table structure.
pypdfium2==5.13.0
pdfplumber==0.11.10

# Regulation graph. Sections, exhibits, definitions and rules as nodes, with the
