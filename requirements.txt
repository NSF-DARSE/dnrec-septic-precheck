# Pinned to the versions this was developed and tested against.
#
# numpy is pinned explicitly even though nothing here imports it directly.
# pandas 1.5.3 ships compiled extensions built against the numpy 1.x C API and
# fails at import with "numpy.dtype size changed" under numpy 2.x. Leaving numpy
# unpinned lets any later install resolve it to 2.x and break pandas, which is
# exactly what happened to the global interpreter on this machine.
numpy==1.23.5

# Upgraded from 1.34.162, which predates Bedrock Data Automation and so exposes no
# bedrock-data-automation client at all. Pure Python, nothing here compiles against
# it, and the resolver leaves numpy alone.
boto3==1.43.73
botocore==1.43.73
requests==2.34.2
pandas==1.5.3

# PDF text extraction. pypdfium2 is the text backend: this regulation uses
# embedded fonts with no usable ToUnicode mapping, so pdfminer (under pdfplumber)
# emits (cid:NNN) markers in place of characters. pdfplumber is kept only for
# extract_tables, since PDFium returns text without table structure.
pypdfium2==5.13.0
pdfplumber==0.11.10

# Regulation graph. Sections, exhibits, definitions and rules as nodes, with the
# containment hierarchy and cross references as edges. In-process, JSON on disk,
# no server.
networkx==3.6.1

# Geospatial screening and figures. All three are pinned to versions built
# against the numpy 1.x C API, for the reason at the top of this file.
#
# This is not hypothetical. Installing matplotlib unpinned resolved it to 3.11.1,
# which requires numpy 2, which upgraded numpy to 2.4.6, which broke pandas 1.5.3
# and took the whole test suite down with it. matplotlib 3.7.5 is the last line
# that works against numpy 1.23. Do not relax these without also moving pandas.
#
# shapely and pyproj rather than geopandas on purpose: geopandas pulls GDAL and
# fiona, which is a much larger compiled surface for no benefit here, since the
# only geometry operations needed are nearest distance and a projection.
matplotlib==3.7.5
shapely==2.0.6
pyproj==3.6.1

PyYAML==6.0.2
pytest==9.1.1

# Reviewer chatbot. Uses the google-genai SDK with Vertex AI (Application
# Default Credentials, no API key). Requires requests>=2.30.0 via the
# google-auth[requests] transitive dependency. Python 3.11 compatible.
# Version 2.18.1 confirmed working with Vertex AI chat sessions.
google-genai==2.18.1

# Annotated correction PDF. Uses pypdf (BSD-3-Clause) for PDF annotation
# without modifying the original document content.
pypdf==6.16.1
